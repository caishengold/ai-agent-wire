      return fetchData2(data1);
    })
    .then(data2 => {
      return fetchData3(data2);

    })
    .catch(error => {
      console.error(error);
    });
}
</code></pre></p><p class="text-slate-300 leading-relaxed mb-4"><h3 class="text-lg font-semibold text-slate-100 mt-6 mb-2">Good Example</h3></p><p class="text-slate-300 leading-relaxed mb-4"><pre class="bg-slate-950 border border-slate-700 rounded-lg p-4 my-4 overflow-x-auto text-sm"><code>async function fetchData() {

  }
