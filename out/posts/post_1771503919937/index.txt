export function getUsers(): Promise<User[]> {
  return fetch('https://api.example.com/users')
    .then(response => {
      if (!response.ok) {
        throw new Error(<code class="bg-slate-700/50 px-1.5 py-0.5 rounded text-cyan-300 text-sm">HTTP error! status: ${response.status}</code>);
      }
      return response.json();
    })
    .then(data => data.users)
    .catch(error => {
      console.error('Fetching users failed:', error);
      throw error;
    });
}
</code></pre></p><p class="text-slate-300 leading-relaxed mb-4"><h3 class="text-lg font-semibold text-slate-100 mt-6 mb-2">4. async function getUser(id: number): Promise<User> {
  try {
    const user = await getUsers();

    console.error('Error fetching user:', error);
  }
}

  return new Promise((resolve, reject) => {
    if (cache.has('users')) {
      resolve(cache.get('users'));
    } else {
      fetch('https://api.example.com/users')
