class PipelineValidator:
    """Validates CI/CD pipeline configurations for common issues."""
    
    def __init__(self, config_path: str):
        self.config_path = Path(config_path)
        self.errors: List[ValidationError] = []
    
    def validate_yaml_syntax(self) -> bool:
        """Check if the YAML file is valid."""
        try:
            with open(self.config_path, 'r') as f:
                yaml.safe_load(f)
            return True
        except yaml.YAMLError as e:
            self.errors.append(ValidationError(
                file=str(self.config_path),
                line=getattr(e, 'problem_mark', None).line if hasattr(e, 'problem_mark') else 0,
                message=f"Invalid YAML syntax: {e}",
                severity='error'
            ))
            return False
    
    def check_required_secrets(self) -> None:
        """Verify required secrets are referenced, not hardcoded."""
        with open(self.config_path, 'r') as f:
            content = f.read()
            lines = content.split('\n')
        
        suspicious_patterns = [
            ('password:', 'Hardcoded password detected'),
            ('api_key:', 'Hardcoded API key detected'),
            ('secret:', 'Potential hardcoded secret'),
            ('AWS_SECRET_ACCESS_KEY=', 'AWS secret in plain text'),
        ]
        
        for line_num, line in enumerate(lines, 1):
            for pattern, message in suspicious_patterns:
                if pattern in line.lower() and '${{' not in line:
                    self.errors.append(ValidationError(
                        file=str(self.config_path),
                        line=line_num,
                        message=message,
                        severity='error'
                    ))
    
    def validate_job_dependencies(self) -> None:
        """Check for circular dependencies and undefined job references."""
        with open(self.config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        if not config or 'jobs' not in config:
            return
        
        jobs = config['jobs']
        job_names = set(jobs.keys())
        
        for job_name, job_config in jobs.items():
            if 'needs' in job_config:
                needed_jobs = job_config['needs']
                if isinstance(needed_jobs, str):
                    needed_jobs = [needed_jobs]
                
                for needed in needed_jobs:
                    if needed not in job_names:
                        self.errors.append(ValidationError(
                            file=str(self.config_path),
                            line=0,
                            message=f"Job '{job_name}' depends on undefined job '{needed}'",
                            severity='error'
                        ))
    
    def run_all_validations(self) -> bool:
        """Run all validation checks and return True if no errors found."""
        print(f"Validating {self.config_path}...")
        
        if not self.validate_yaml_syntax():
            return False
        
        self.check_required_secrets()
        self.validate_job_dependencies()
        
        # Filter to only errors for the return value
        errors_only = [e for e in self.errors if e.severity == 'error']
        
        if errors_only:
            print(f"\nFound {len(errors_only)} error(s):")
            for error in errors_only:
                print(f"  ❌ Line {error.line}: {error.message}")
        
        warnings = [e for e in self.errors if e.severity == 'warning']
        if warnings:
            print(f"\nFound {len(warnings)} warning(s):")
            for warning in warnings:
                print(f"  ⚠️  Line {warning.line}: {warning.message}")
        
        if not errors_only and not warnings:
            print("  ✓ All validations passed")
        
        return len(errors_only) == 0</p><p class="text-slate-300 leading-relaxed mb-4">

    """Main entry point for the validator."""
    import argparse
    
    parser = argparse.ArgumentParser(

    parser.add_argument(

Use Infrastructure as Code (Terraform, CloudFormation, or Pulumi) to ensure environments are identical except for scale and credentials.</p><p class="text-slate-300 leading-relaxed mb-4"></strong>Secrets belong in vaults, not repositories.<strong> Hardcoded credentials in pipeline files are a data breach waiting to happen. Use your platform's secret management features and rotate credentials regularly. The validator script above catches accidental hardcoding before it reaches your repository.</p><p class="text-slate-300 leading-relaxed mb-4"></strong>Speed matters for feedback loops.<strong> Pipelines that take 45 minutes encourage developers to batch changes and skip runs. Aim for sub-10 minute feedback on pull requests through parallelisation, intelligent test splitting, and caching. Slow pipelines are broken pipelines.</p><p class="text-slate-300 leading-relaxed mb-4"></strong>Monitoring applies to pipelines too.<strong> Track pipeline duration, success rates, and queue times. Alert when success rates drop below 95% or when queue times exceed 10 minutes. A failing pipeline is a production risk, not just a developer inconvenience.</p><p class="text-slate-300 leading-relaxed mb-4"><h2 class="text-xl font-bold text-slate-100 mt-8 mb-3">Actionable Takeaways</h2></p><p class="text-slate-300 leading-relaxed mb-4"><li class="ml-4 list-decimal text-slate-300"></strong>Audit exit code handling in your current pipelines<strong> — Add <code class="bg-slate-700/50 px-1.5 py-0.5 rounded text-cyan-300 text-sm">set -e</code> to all shell scripts and verify that failed commands actually fail the pipeline. Check the last three incidents for exit code issues.</li></p><p class="text-slate-300 leading-relaxed mb-4"><li class="ml-4 list-decimal text-slate-300"></strong>Implement mandatory security scanning<strong> — Add dependency vulnerability scanning (Snyk, Dependabot, or Safety) and secret detection (GitGuardian or TruffleHog) to every pipeline before the deploy stage.</li></p><p class="text-slate-300 leading-relaxed mb-4"><li class="ml-4 list-decimal text-slate-300"></strong>Measure pipeline feedback time<strong> — Track the time from commit to test results for pull requests. If it exceeds 10 minutes, identify the slowest stage and optimise or parallelise it.</li></p><p class="text-slate-300 leading-relaxed mb-4"><li class="ml-4 list-decimal text-slate-300"></strong>Create environment parity with Infrastructure as Code<strong> — Ensure staging and production use identical configurations. Any drift should be treated as a bug and fixed immediately.</li></p><p class="text-slate-300 leading-relaxed mb-4"><li class="ml-4 list-decimal text-slate-300"></strong>Set up pipeline monitoring** — Add alerts for pipeline success rates, queue times, and duration trends. Treat pipeline degradation with the same urgency as production alerts.</li></p><p class="text-slate-300 leading-relaxed mb-4">Well-designed CI/CD pipelines make deployments boring.
