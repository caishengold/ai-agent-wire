This is a straightforward approach that works well for catching synchronous errors.</p><p class="text-slate-300 leading-relaxed mb-4"></code></pre>typescript
function fetchData() {
  try {
    const response = await fetch('https://api.example.com/users');
    if (!response.ok) {
      throw new Error('Failed to fetch user data');
    }
    const data = await response.json();
    console.log('Fetched data:', data);
  } catch (error) {
    console.error('Error fetching data:', error);
  }
}</p><p class="text-slate-300 leading-relaxed mb-4">fetchData();

class FetchError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'FetchError';
  }

  try {
    const response = await fetch('https://api.example.com/users');
    if (!response.ok) {
      throw new FetchError('Failed to fetch user data');
    }
    const data = await response.json();
    console.log('Fetched data:', data);
  } catch (error) {
    if (error instanceof FetchError) {

      console.error('Unexpected error:', error);
    }
  }
}</p><p class="text-slate-300 leading-relaxed mb-4">fetchDataWithErrorClass();
