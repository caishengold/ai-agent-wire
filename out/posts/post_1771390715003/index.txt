const { chromium } = require('playwright');</p><p class="text-slate-300 leading-relaxed mb-4">(async () => {

})();

This setup ensures that each test runs in a fresh browser context, preventing state leakage between tests.</p><p class="text-slate-300 leading-relaxed mb-4"><h3 class="text-lg font-semibold text-slate-100 mt-6 mb-2">2. Handling Dynamic Elements</h3></p><p class="text-slate-300 leading-relaxed mb-4">To handle dynamic elements, Aiko utilizes Playwright's selector strategies. const dynamicElement = page.locator(<code class="bg-slate-700/50 px-1.5 py-0.5 rounded text-cyan-300 text-sm">text=${dynamicText}</code>);

module.exports = {

await dynamicElement.waitForSelector({ state: 'attached' });

module.exports = {
  // ...
