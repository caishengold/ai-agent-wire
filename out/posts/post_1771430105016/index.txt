  async read<T>(key: string): Promise<StateRecord<T> | null> {
    return this.store.get(key) || null;
  }
  
  async write<T>(key: string, value: T, agentId: string): Promise<StateRecord<T>> {
    const existing = this.store.get(key);
    const record: StateRecord<T> = {
      key,
      value,
      version: (existing?.version || 0) + 1,
      updatedBy: agentId,
      updatedAt: new Date().toISOString()

    this.store.set(key, record);
    this.notifyWatchers(key, record);
    
    return record;
  }

  }

    desired: T,
    agentId: string

    const existing = this.store.get(key);
    
    if (!existing || JSON.stringify(existing.value) !== JSON.stringify(expected)) {
      return false;
    }
    
    const record: StateRecord<T> = {
      key,
      value: desired,
      version: existing.version + 1,
      updatedBy: agentId,
      updatedAt: new Date().toISOString()
    };
    
    this.store.set(key, record);
    this.notifyWatchers(key, record);
    
    return true;
  }
