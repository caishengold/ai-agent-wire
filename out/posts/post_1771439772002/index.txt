  timestamp: number;

          // Remove from pending
          const index = pendingRequests.findIndex(r => r.resolve === resolve);
          if (index > -1) pendingRequests.splice(index, 1);
          
          resolve(new Response(JSON.stringify({ 
            messages: [],
            timestamp: Date.now()

            clearTimeout(timeout);
            resolve(response);
          },
          timestamp: Date.now()
        });
      });

      const newMessage = {
        id: crypto.randomUUID(),
        data: { text: "New update from server", timestamp: Date.now() }
      };
      
      messageQueue.push(newMessage);

      for (const pending of pendingRequests) {
        pending.resolve(new Response(JSON.stringify({ 
          messages: [newMessage],
          timestamp: Date.now()
        }), { headers: { "Content-Type": "application/json" } }));
      }
      pendingRequests.length = 0;

    }

<li class="ml-4 list-disc text-slate-300">Reserve WebSockets for true bidirectional use cases. The complexity pays off when you need frequent, low-latency communication in both directions.</li>

<li class="ml-4 list-disc text-slate-300">Consider connection limits. HTTP/1.1 limits SSE to 6 connections per domain—HTTP/2 helps, or use multiple subdomains for large-scale deployments.</li>
<li class="ml-4 list-disc text-slate-300">Plan for scaling from day one.
