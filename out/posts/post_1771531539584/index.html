TL;DR: This guide provides copy-paste ready TypeScript code and JSON schemas for implementing function calling across OpenAI, Anthropic Claude, and Google Gemini. Each provider uses distinct formats for tool definitions, tool choice control, and error handling.

1. OpenAI Tool Choice Format

Tool Definition Schema

interface OpenAIToolDefinition {
  type: "function";
  function: {
    name: string;
    description: string;
    parameters: OpenAIJSONSchema;
    strict?: boolean;
  };
}
interface OpenAIJSONSchema {
  type: "object";
  properties: Record;
  required?: string[];
  [key: string]: unknown;
}

Tool Choice Configuration

type OpenAIToolChoice = 
  | { type: "auto" }
  | { type: "none" }
  | { type: "required" }
  | { type: "function"; function: { name: string } }
  | { type: "allowed_tools"; tools: Array<{ type: "function"; name: string }> };

JSON Schema Examples

{
  "tools": [
    {
      "type": "function",
      "function": {
        "name": "get_weather",
        "description": "Get current weather for a location",
        "parameters": {
          "type": "object",
          "properties": {
            "location": {
              "type": "string",
              "description": "City name"
            },
            "unit": {
              "type": "string",
              "enum": ["celsius", "fahrenheit"]
            }
          },
          "required": ["location"]
        },
        "strict": true
      }
    }
  ],
  "tool_choice": {
    "type": "function",

      "name": "get_weather"
    }
  }
}

TypeScript Implementation

import OpenAI from "openai";
const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
const tools: OpenAI.Chat.Completions.ChatCompletionTool[] = [
  {
    type: "function",
    function: {
      name: "get_weather",
      description: "Get current weather for a location",
      parameters: {
        type: "object",
        properties: {
          location: { type: "string", description: "City name" },
          unit: { type: "string", enum: ["celsius", "fahrenheit"] }
        },
        required: ["location"]
      }
    }
  }
];
const response = await client.chat.completions.create({
  model: "gpt-4o",
  messages: [{ role: "user", content: "What's the weather in Tokyo?" }],
  tools,
  tool_choice: { type: "auto" }
});
const toolCall = response.choices[0].message.tool_calls?.[0];
if (toolCall) {
  const args = JSON.parse(toolCall.function.arguments);
  console.log("Function:", toolCall.function.name);
  console.log("Arguments:", args);
}

TypeScript Implementation

import Anthropic from "@anthropic-ai/sdk";
const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });
const tools: Anthropic.Tools.Tool[] = [
  {
    name: "get_weather",
    description: "Get current weather for a location",
    input_schema: {
      type: "object",
      properties: {
        location: { type: "string", description: "City name" },
        unit: { type: "string", enum: ["celsius", "fahrenheit"], description: "Temperature unit" }
      },
      required: ["location"]
    }
  }
];
const message = await client.messages.create({
  model: "claude-sonnet-4-20250514",
  max_tokens: 1024,
  tools,
  tool_choice: { type: "auto" },
  messages: [{ role: "user", content: "What's the weather in Tokyo?" }]
});
const toolUse = message.content.find((block): block is Anthropic.ToolUseBlock => 
  block.type === "tool_use"
);
if (toolUse) {
  console.log("Tool:", toolUse.name);
  console.log("Input:", toolUse.input);
  console.log("Tool ID:", toolUse.id);
}
const result = await client.messages.create({
  model: "claude-sonnet-4-20250514",
  max_tokens: 1024,
  tools,
  messages: [
    { role: "user", content: "What's the weather in Tokyo?" },
    message.content[0] as Anthropic.MessageParam,
    {
      type: "tool_result",
      tool_use_id: toolUse?.id,
      content: JSON.stringify({ temperature: 22, condition: "sunny" })
    }
  ]
});

TypeScript Implementation

import { GoogleGenerativeAI } from "@google/generative-ai";
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);
const model = genAI.getGenerativeModel({
  model: "gemini-2.0-flash",
  tools: [
    {
      functionDeclarations: [
        {
          name: "get_weather",
          description: "Get current weather for a location",
          parameters: {
            type: "object",
            properties: {
              location: { type: "string", description: "City name" },
              unit: { type: "string", enum: ["celsius", "fahrenheit"], description: "Temperature unit" }
            },
            required: ["location"]
          }
        }
      ]
    }
  ],
  toolConfig: {
    functionCallingConfig: {
      mode: "AUTO"
    }
  }
});
const chat = model.startChat({
  history: [{ role: "user", parts: [{ text: "What's the weather in Tokyo?" }] }]
});
const result = await chat.sendMessage("What's the weather in Tokyo?");
const response = result.response;
if (response.functionCalls && response.functionCalls.length > 0) {
  const functionCall = response.functionCalls[0];
  console.log("Function:", functionCall.name);
  console.log("Arguments:", functionCall.args);
}
const functionResponse = {
  name: functionCall!.name,
  response: {
    temperature: 22,
    condition: "sunny"
  }
};
const finalResult = await chat.sendMessage([{ functionResponse }]);
console.log(finalResult.response.text());

5. Error Handling Patterns

OpenAI Error Handling

interface OpenAIFunctionCallResult {
  name: string;
  arguments: string;
}
function handleOpenAIToolCall(toolCall: OpenAI.Chat.Completions.ChatCompletionMessageToolCall) {
  try {
    const args = JSON.parse(toolCall.function.arguments);
    
    if (!args.location) {
      return {
        role: "tool" as const,
        tool_call_id: toolCall.id,
        content: JSON.stringify({ error: "Missing required parameter: location" })
      };
    }
    
    const result = executeWeatherTool(args);
    return {
      role: "tool" as const,
      tool_call_id: toolCall.id,
      content: JSON.stringify(result)
    };
  } catch (error) {
    if (error instanceof SyntaxError) {
      return {
        role: "tool" as const,
        tool_call_id: toolCall.id,
        content: JSON.stringify({ error: "Invalid JSON arguments" })
      };
    }
    throw error;
  }
}
async function openAIFunctionCallLoop(
  client: OpenAI,
  model: string,
  messages: OpenAI.Chat.Completions.ChatMessage[],
  tools: OpenAI.Chat.Completions.ChatCompletionTool[]
): Promise {
  const maxIterations = 10;
  
  for (let i = 0; i < maxIterations; i++) {
    const response = await client.chat.completions.create({
      model,
      messages,
      tools,
      tool_choice: { type: "auto" }
    });
    
    const toolCalls = response.choices[0].message.tool_calls;
    
    if (!toolCalls || toolCalls.length === 0) {
      return response.choices[0].message.content || "";
    }
    
    const toolResults = toolCalls.map(handleOpenAIToolCall);
    messages.push({
      role: "assistant",
      content: response.choices[0].message.content
    } as any);
    messages.push(...toolResults);
  }
  
  throw new Error("Max iterations exceeded");
}

Anthropic Error Handling

type AnthropicToolResultContent = 
  | { type: "tool_result"; tool_use_id: string; content: string; }
  | { type: "tool_result"; tool_use_id: string; content: string; is_error: boolean; };
function handleAnthropicToolUse(toolUse: Anthropic.ToolUseBlock): AnthropicToolResultContent {
  try {
    const input = toolUse.input;
    
    if (!("location" in input) || !input.location) {
      return {
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify({ error: "Missing required parameter: location" }),
        is_error: true
      };
    }
    
    const result = executeWeatherTool(input);
    return {
      type: "tool_result",
      tool_use_id: toolUse.id,
      content: JSON.stringify(result)
    };
  } catch (error) {
    return {
      type: "tool_result",
      tool_use_id: toolUse.id,
      content: JSON.stringify({ error: String(error) }),
      is_error: true
    };
  }
}
async function anthropicToolUseLoop(
  client: Anthropic,
  model: string,
  system: string,
  messages: Anthropic.MessageParam[]
): Promise {
  const maxIterations = 10;
  
  for (let i = 0; i < maxIterations; i++) {
    const message = await client.messages.create({
      model,
      max_tokens: 4096,
      system,
      tools,
      tool_choice: { type: "auto" },
      messages
    });
    
    const toolUses = message.content.filter((b): b is Anthropic.ToolUseBlock => 
      b.type === "tool_use"
    );
    
    if (toolUses.length === 0) {
      return message.content
        .filter((b): b is Anthropic.TextBlock => b.type === "text")
        .map(b => b.text)
        .join("");
    }
    
    const toolResults = toolUses.map(handleAnthropicToolUse);
    messages.push({ role: "user", content: message.content as any });
    messages.push(...toolResults);
  }
  
  throw new Error("Max iterations exceeded");
}

Google Gemini Error Handling

interface GeminiFunctionCall {
  name: string;
  args: Record;
}
function handleGeminiFunctionCall(functionCall: GeminiFunctionCall) {
  try {
    const { name, args } = functionCall;
    
    if (!args.location) {
      return {
        name,
        response: { error: "Missing required parameter: location" }
      };
    }
    
    const result = executeWeatherTool(args);
    return { name, response: result };
  } catch (error) {
    return {
      name: functionCall.name,
      response: { error: String(error) }
    };
  }
}
async function geminiFunctionCallLoop(
  model: GenerativeModel,
  prompt: string
): Promise {
  const chat = model.startChat();
  const maxIterations = 10;
  
  for (let i = 0; i < maxIterations; i++) {
    const result = await chat.sendMessage(prompt);
    const response = result.response;
    
    const functionCalls = response.functionCalls;
    if (!functionCalls || functionCalls.length === 0) {
      return response.text();
    }
    
    const functionResponses = functionCalls.map(handleGeminiFunctionCall);
    const parts = functionResponses.map(fr => ({ functionResponse: fr }));
    
    const nextResult = await chat.sendMessage(parts);
    prompt = "";
  }
  
  throw new Error("Max iterations exceeded");
}

6. Provider Conversion Utilities

function openAItoAnthropicTool(tool: OpenAI.Chat.Completions.ChatCompletionTool): Anthropic.Tools.Tool {
  const fn = tool.function;
  return {
    name: fn.name,
    description: fn.description,
    input_schema: fn.parameters as AnthropicJSONSchema
  };
}
function anthropicToOpenAITool(tool: Anthropic.Tools.Tool): OpenAI.Chat.Completions.ChatCompletionTool {
  return {
    type: "function",
    function: {
      name: tool.name,
      description: tool.description,
      parameters: tool.input_schema as any
    }
  };
}
function openAItoGeminiTool(tool: OpenAI.Chat.Completions.ChatCompletionTool): GeminiTool {
  return {
    functionDeclarations: [{
      name: tool.function.name,
      description: tool.function.description,
      parameters: tool.function.parameters as GeminiOpenAPISchema
    }]
  };
}
function geminiToOpenAITool(tool: GeminiTool): OpenAI.Chat.Completions.ChatCompletionTool {
  const decl = tool.functionDeclarations[0];
  return {
    type: "function",
    function: {
      name: decl.name,
      description: decl.description,
      parameters: decl.parameters as any
    }
  };
}
